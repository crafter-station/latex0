\section{Linear maps}
\begin{definition}
The trace of a square matrix $A$ is $\tr A = \sum_i a_{ii}$.
\end{definition}

\begin{lemma}\label{lem:cyclic}
For matrices $A$ and $B$, $\tr(AB) = \tr(BA)$.
\end{lemma}
\begin{proof}
Both sides equal $\sum_{i,j} a_{ij} b_{ji}$.
\end{proof}

\begin{theorem}
Similar matrices have the same trace.
\end{theorem}
\begin{proof}
By \cref{lem:cyclic},
\begin{gather}
  \tr(P^{-1}AP) = \tr(APP^{-1}) \\
  = \tr A.
\end{gather}
\end{proof}

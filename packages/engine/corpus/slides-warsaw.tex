\section{Motivation}
\begin{frame}{Why monitor soil moisture?}
  \begin{itemize}
    \item Agriculture uses about 70\% of global freshwater.
    \item Over-irrigation leaches nutrients and wastes energy.
    \item Sensor networks enable irrigation on demand.
  \end{itemize}
\end{frame}

\section{System design}
\begin{frame}{Node architecture}
  \begin{description}
    \item[MCU] ARM Cortex-M0+, 48 MHz
    \item[Radio] LoRa, 868 MHz, SF7--SF12
    \item[Sensors] capacitive moisture, temperature
    \item[Power] 2 $\times$ AA cells, solar trickle charge
  \end{description}
\end{frame}

\section{Conclusion}
\begin{frame}{Summary}

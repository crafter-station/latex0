\usepackage[ngerman,french,english]{babel}
\begin{document}

\end{otherlanguage}

\end{otherlanguage*}

\begin{otherlanguage}{ngerman}

\end{otherlanguage}
Encore le \today.

\selectlanguage{english}
Again \today.

\documentclass{article}
\usepackage{url}
\newcounter{q}
\newenvironment{answer}{\par\noindent\textbf{Answer.}\ }{\par}
\begin{document}
\stepcounter{q}Question \theq.
\begin{answer}
It is \[ \sum_i x_i^2 \] for all $x$.
\end{answer}
\begin{answer}
See question \theq{} again: it costs \$5 and $\alpha$.
\end{answer}
\begin{answer}
At \url{https://example.org} here.\footnote{A note.}
\end{answer}
\end{document}

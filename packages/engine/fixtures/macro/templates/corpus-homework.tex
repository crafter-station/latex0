\section*{Problem 1}
Show that $\sum_{k=1}^{n} k = \frac{n(n+1)}{2}$.

\textbf{Proof.} By induction on $n$. For $n=1$ both sides equal $1$. Assume it
holds for $n$; then
\begin{align*}
  \sum_{k=1}^{n+1} k &= \frac{n(n+1)}{2} + (n+1) \\
                     &= \frac{(n+1)(n+2)}{2}.
\end{align*}

\section*{Problem 2}
Solve the system
\begin{enumerate}
  \item $x + y = 3$,
  \item $x - y = 1$.
\end{enumerate}
Adding the equations gives $x = 2$, so $y = 1$.
